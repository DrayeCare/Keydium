#include "core/KeySimulator.h"

#ifdef _WIN32
    #include <windows.h>
#elif __linux__
    #include <X11/Xlib.h>
    #include <X11/extensions/XTest.h>
    #include <X11/keysym.h>
    // Static display pointer agar tidak buka-tutup koneksi X11 terus menerus
    static Display* display = nullptr; 
#endif

namespace Core {

    void KeySimulator::PressKey(int keyCode) {
        #ifdef _WIN32
            // keyCode di Windows adalah Virtual-Key Code (misal 0x41 untuk 'A')
            INPUT input = {0};
            input.type = INPUT_KEYBOARD;
            input.ki.wVk = keyCode;
            SendInput(1, &input, sizeof(INPUT));
            
        #elif __linux__
            // keyCode di Linux sebaiknya menggunakan X11 KeySym (misal XK_a)
            if (!display) display = XOpenDisplay(nullptr);
            if (display) {
                KeyCode kc = XKeysymToKeycode(display, keyCode);
                XTestFakeKeyEvent(display, kc, True, 0); // True = Tekan
                XFlush(display);
            }
        #endif
    }

    void KeySimulator::ReleaseKey(int keyCode) {
        #ifdef _WIN32
            INPUT input = {0};
            input.type = INPUT_KEYBOARD;
            input.ki.wVk = keyCode;
            input.ki.dwFlags = KEYEVENTF_KEYUP; // Flag untuk lepas
            SendInput(1, &input, sizeof(INPUT));
            
        #elif __linux__
            if (display) {
                KeyCode kc = XKeysymToKeycode(display, keyCode);
                XTestFakeKeyEvent(display, kc, False, 0); // False = Lepas
                XFlush(display);
            }
        #endif
    }
}

PianoKeyboard::PianoKeyboard() {
    setOpaque(false);
    // Contoh mapping awal (bisa diubah nanti)
    mappedKeys = { 48, 50, 52, 53, 55, 57, 59 }; // C3, D3, E3, F3, G3, A3, B3
    rebuildKeys();
}

PianoKeyboard::~PianoKeyboard() {}

void PianoKeyboard::resized() { rebuildKeys(); }

void PianoKeyboard::setStartNote(int note) {
    startNote = note;
    rebuildKeys();
    repaint();
}

void PianoKeyboard::setNumWhiteKeys(int count) {
    numWhiteKeys = count;
    rebuildKeys();
    repaint();
}

void PianoKeyboard::setMappedKeys(const std::set<int>& keys) {
    mappedKeys = keys;
    repaint();
}

void PianoKeyboard::setActiveKeys(const std::set<int>& keys) {
    activeKeys = keys;
    repaint();
}

bool PianoKeyboard::isBlackKey(int note) {
    int n = note % 12;
    return n == 1 || n == 3 || n == 6 || n == 8 || n == 10;
}

juce::String PianoKeyboard::getNoteName(int note) {
    static const char* names[] = {"C","C#","D","D#","E","F","F#","G","G#","A","A#","B"};
    int octave = (note / 12) - 1;
    return juce::String(names[note % 12]) + juce::String(octave);
}

void PianoKeyboard::rebuildKeys() {
    keys.clear();
    auto area = getLocalBounds().toFloat();
    if (area.getWidth() < 10) return;
    
    float whiteKeyW = area.getWidth() / (float)numWhiteKeys;
    float blackKeyW = whiteKeyW * 0.62f;
    float blackKeyH = area.getHeight() * 0.62f;
    
    // Cari C terdekat dari startNote
    int firstC = startNote - (startNote % 12);
    
    // Gambar white keys
    int whiteIdx = 0;
    for (int n = firstC; whiteIdx < numWhiteKeys; n++) {
        if (isBlackKey(n)) continue;
        KeyInfo k;
        k.note = n;
        k.isBlack = false;
        k.bounds = { area.getX() + whiteIdx * whiteKeyW,
                     area.getY(),
                     whiteKeyW - 1.0f,
                     area.getHeight() };
        keys.push_back(k);
        whiteIdx++;
    }
    
    // Gambar black keys
    for (int n = firstC; n < firstC + numWhiteKeys * 2; n++) {
        if (!isBlackKey(n)) continue;
        int prevWhite = n - 1;
        for (auto& wk : keys) {
            if (!wk.isBlack && wk.note == prevWhite) {
                KeyInfo k;
                k.note = n;
                k.isBlack = true;
                k.bounds = { wk.bounds.getRight() - blackKeyW * 0.5f,
                             area.getY(),
                             blackKeyW,
                             blackKeyH };
                keys.push_back(k);
                break;
            }
        }
    }
}

void PianoKeyboard::paint(juce::Graphics& g) {
    // Draw white keys
    for (auto& k : keys) {
        if (k.isBlack) continue;
        bool mapped = mappedKeys.count(k.note) > 0;
        bool active = activeKeys.count(k.note) > 0;
        
        // Key body
        g.setColour(active ? juce::Colour(0xffffeb3b) : juce::Colours::white);
        g.fillRect(k.bounds);
        g.setColour(juce::Colour(0xff333344));
        g.drawRect(k.bounds, 1.0f);
        
        // Note name (top)
        g.setColour(juce::Colours::black);
        g.setFont(juce::Font(11.0f, juce::Font::bold));
        g.drawText(getNoteName(k.note),
                   k.bounds.reduced(2).removeFromTop(20),
                   juce::Justification::centredTop);
        
        // Mapped badge (orange)
        if (mapped) {
            auto badge = juce::Rectangle<float>(
                k.bounds.getX() + 2, 
                k.bounds.getBottom() - 40,
                k.bounds.getWidth() - 4, 
                16);
            g.setColour(juce::Colour(0xffffa726));
            g.fillRoundedRectangle(badge, 3.0f);
            g.setColour(juce::Colours::black);
            g.setFont(juce::Font(10.0f, juce::Font::bold));
            g.drawText("KEY", badge, juce::Justification::centred);
        }
        
        // Note number (bottom)
        g.setColour(juce::Colour(0xff666677));
        g.setFont(juce::Font(9.0f));
        g.drawText(juce::String(k.note),
                   k.bounds.reduced(2).removeFromBottom(15),
                   juce::Justification::centred);
    }
    
    // Draw black keys on top
    for (auto& k : keys) {
        if (!k.isBlack) continue;
        bool mapped = mappedKeys.count(k.note) > 0;
        bool active = activeKeys.count(k.note) > 0;
        
        g.setColour(active ? juce::Colour(0xffffeb3b) : juce::Colour(0xff1a1a1a));
        g.fillRoundedRectangle(k.bounds, 2.0f);
        
        // Note name
        g.setColour(juce::Colours::white);
        g.setFont(juce::Font(9.0f, juce::Font::bold));
        g.drawText(getNoteName(k.note),
                   k.bounds.reduced(1).removeFromTop(15),
                   juce::Justification::centredTop);
    }
}
